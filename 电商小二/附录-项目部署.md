# 一、Ubuntu Docker 安装

> 系统：img‑mmytdhbn | Ubuntu Server 24.04 LTS 64 位

## 1 一键安装脚本

```
#!/bin/bash
set -e

echo "===== 0.清理旧错误源文件 ====="
sudo rm -f /etc/apt/sources.list.d/docker.sources
sudo rm -f /etc/apt/sources.list.d/docker.list

echo "===== 1.清理旧冲突包 ====="
sudo apt update
sudo apt remove -y docker docker.io docker-doc podman-docker containerd runc

echo "===== 2.安装依赖，导入阿里云Docker GPG密钥 ====="
sudo apt install -y ca-certificates curl
sudo install -m 0755 -d /etc/apt/keyrings
sudo curl -fsSL https://mirrors.aliyun.com/docker-ce/linux/ubuntu/gpg -o /etc/apt/keyrings/docker.asc
sudo chmod a+r /etc/apt/keyrings/docker.asc

# 提前获取系统代号与架构，避免heredoc变量解析失败
CODENAME=$(. /etc/os-release && echo "${UBUNTU_CODENAME:-$VERSION_CODENAME}")
ARCH=$(dpkg --print-architecture)
echo ">> 识别系统: $CODENAME , 架构: $ARCH"

echo "===== 3.写入阿里云docker apt源 ====="
sudo tee /etc/apt/sources.list.d/docker.sources >/dev/null <<EOF
Types: deb
URIs: https://mirrors.aliyun.com/docker-ce/linux/ubuntu
Suites: ${CODENAME}
Components: stable
Architectures: ${ARCH}
Signed-By: /etc/apt/keyrings/docker.asc
EOF

# 校验源文件内容
echo ">> 检查生成的源文件内容："
cat /etc/apt/sources.list.d/docker.sources

echo "===== 4.安装Docker全套组件 ====="
sudo apt update
sudo apt install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin

echo "===== 5.启动并设置开机自启 ====="
sudo systemctl enable --now docker
sudo systemctl status docker --no-pager

echo "===== 6.配置国内镜像加速器 ====="
sudo mkdir -p /etc/docker
sudo tee /etc/docker/daemon.json <<-'EOF'
{
  "registry-mirrors": [
    "https://docker.1panel.live",
    "https://docker.xuanyuan.me",
    "https://docker.1ms.run",
    "https://docker.m.daocloud.io"
  ]
}
EOF

sudo systemctl daemon-reload
sudo systemctl restart docker

echo "===== 7.配置docker用户组免sudo（SSH重连生效） ====="
sudo usermod -aG docker $USER

echo ""
echo "=============================================="
echo "Docker 安装完成！"
echo "注意：免sudo权限需要断开SSH重新登录才生效！"
echo ""
echo "验证命令："
echo "  docker --version"
echo "  docker compose version"
echo "  docker info"
echo "  docker run hello-world"
echo "=============================================="
```


## 2 两种运行方式

### 方式 1：直接粘贴全部内容到终端回车执行

### 方式 2：保存脚本文件执行

```
vim install-docker.sh
# 将上面脚本全部粘贴进去保存退出
chmod +x install-docker.sh
./install-docker.sh
```

## 3 安装完之后

> **必须断开 SSH，重新连接服务器，`docker`不用 sudo 才能生效。**

测试命令：

```
docker run hello-world
```

## 4 原始方式部署

* 安装python和uv环境

```shell
sudo apt update
sudo apt install python3-pip -y

pip3 install uv --user --break-system-packages

# 然后配置PATH
echo 'export PATH="$HOME/.local/bin:$PATH"' >> ~/.bashrc

source ~/.bashrc

#export UV_DEFAULT_INDEX=https://pypi.tuna.tsinghua.edu.cn/simple

# 当前终端会话生效
export UV_DEFAULT_INDEX="https://mirrors.aliyun.com/pypi/simple/"

uv sync

# 如果uv不能使用
# 手动临时补path测试
# export PATH="$HOME/.local/bin:$PATH"
uv --version
```

* 启动服务

```
uv run python main.py
```



## 4 新建 Dockerfile

* 在项目根目录下创建
* 文件名称是 Dockerfile （无后缀名）

```
FROM ghcr.m.daocloud.io/astral-sh/uv:python3.12-bookworm-slim

WORKDIR /app

# 直接拷贝本地的虚拟环境，不执行任何uv sync，不下载依赖
COPY .venv ./.venv
COPY pyproject.toml uv.lock ./
# RUN uv lock && uv sync --no-cache --no-dev
COPY . .

EXPOSE 18081
CMD ["uv", "run", "python", "main.py"]
```



## 5 一次性部署步骤（按顺序执行）

### 5.1  预拉取镜像（防止构建超时）

```shell
docker pull ghcr.m.daocloud.io/astral-sh/uv:python3.12-bookworm-slim
```

### 5.2 构建镜像（末尾小数点必须有）

```shell
docker build --progress=plain -t ecommerce-backend:v1 .
```

### 5.3 启动容器

```shell
docker run -d \
  --name ecommerce-backend \
  -p 18081:18081 \
  --restart always \
  ecommerce-backend:v1
```
